cmake_minimum_required(VERSION 3.23)
project(parallel_ndd VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PNDD_LTO "Interprocedural optimization for Release builds" ON)
if(PNDD_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT pndd_ipo_available OUTPUT pndd_ipo_error)
  if(NOT pndd_ipo_available)
    message(FATAL_ERROR "PNDD_LTO is unavailable: ${pndd_ipo_error}. Use -DPNDD_LTO=OFF.")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
include(CTest)
option(PNDD_BUILD_JNI "Build the Java binding" ON)
option(PNDD_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(PNDD_PROFILE "Diagnostic timers and task histograms (perturbs timings)" OFF)
option(PNDD_PAR_PROFILE "Sample parallel lock waits and worker task load (perturbs timings)" OFF)
option(PNDD_OPTIMISTIC_CACHE "Atomic versioned operation cache with nonblocking readers/writers" ON)
option(PNDD_BATCH_GRAIN "Prefer outstanding coarse batch work over nested fine tasks" OFF)
option(PNDD_PENDING_PARTITION "Split only unresolved NDD child jobs" OFF)
option(PNDD_LABEL_GRAIN "Spawn label branches only when both have uncached work" OFF)
option(PNDD_ROOT_AFFINITY "Keep serialized external RUN tasks on a warm coordinating worker" ON)
option(PNDD_FAST_BCDD "Specialized checked BCDD AND kernel" ON)
option(PNDD_SPLIT_CACHE "Separate NDD and label regions within the configured cache budget" ON)
option(PNDD_PAD_ROOTS "Isolate worker root registries on separate cache lines" ON)
option(PNDD_LOCAL_CACHE "Worker-local front cache for recursive operations" ON)
option(PNDD_LEAF_APPLY "Apply labels directly when single-edge nodes share a child" ON)
option(PNDD_RESIDUAL_APPLY "Prune ordered OR/DIFF edge residuals without coverage unions" ON)
option(PNDD_LABEL_GC_KEEP_NDD "Preserve NDD memo entries during label-only GC" ON)
option(PNDD_NODE_SLABS "Cache-line aligned slab storage for terminal and single-edge nodes" ON)
option(PNDD_COST_GRAIN "Resolve trivial/cached children before deciding to spawn" ON)
set(PNDD_SYLVAN_INITIAL_RATIO 2 CACHE STRING "Initial Sylvan capacity divisor exponent for budgets above 128 MiB")
if(NOT PNDD_SYLVAN_INITIAL_RATIO MATCHES "^[0-8]$")
  message(FATAL_ERROR "PNDD_SYLVAN_INITIAL_RATIO must be 0..8")
endif()
option(PNDD_EXPERIMENT_SATCOUNT_CACHE "Persistent per-node NDD count cache (legacy option name)" ON)
option(PNDD_EXPERIMENT_BCDD_AND "Experimental shared AND-only BCDD label kernel" OFF)
option(PNDD_EXPERIMENT_SYLVAN_CACHE "Experimental Sylvan lock-free BCDD apply cache" OFF)
set(PNDD_SYLVAN_SOURCE "" CACHE PATH "Local checkout of the pinned Sylvan commit (optional)")
set(PNDD_SYLVAN_COMMIT c8bf561ee64036375f4693eb389e57ec8ef4aa6d)
if(PNDD_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_sylvan.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_lace.py")
if(PNDD_SYLVAN_SOURCE)
  execute_process(COMMAND git -C "${PNDD_SYLVAN_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE sylvan_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT sylvan_revision STREQUAL PNDD_SYLVAN_COMMIT)
    message(FATAL_ERROR "PNDD_SYLVAN_SOURCE must be at ${PNDD_SYLVAN_COMMIT}")
  endif()
  set(sylvan_source "${PNDD_SYLVAN_SOURCE}")
else()
  include(FetchContent)
  FetchContent_Declare(pndd_sylvan GIT_REPOSITORY https://github.com/Augists/sylvan.git
    GIT_TAG ${PNDD_SYLVAN_COMMIT})
  FetchContent_GetProperties(pndd_sylvan)
  if(NOT pndd_sylvan_POPULATED)
    FetchContent_Populate(pndd_sylvan)
  endif()
  set(sylvan_source "${pndd_sylvan_SOURCE_DIR}")
endif()
# Patch a build-local copy only. A bounded native allocation must not exit the JVM.
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_sylvan.py"
  "${sylvan_source}" "${CMAKE_CURRENT_BINARY_DIR}/sylvan-adapter" COMMAND_ERROR_IS_FATAL ANY)
set(SYLVAN_BUILD_CPP OFF CACHE BOOL "" FORCE)
set(SYLVAN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SYLVAN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SYLVAN_ENABLE_PIC ON CACHE BOOL "" FORCE)
set(LACE_ENABLE_PIC ON CACHE BOOL "" FORCE)
include(FetchContent)
set(PNDD_LACE_COMMIT 63d8a5ea79922a234e284ee4b264fa5e4e784939)
FetchContent_Declare(lace GIT_REPOSITORY https://github.com/trolando/lace.git
  GIT_TAG ${PNDD_LACE_COMMIT})
FetchContent_GetProperties(lace)
if(NOT lace_POPULATED)
  FetchContent_Populate(lace)
endif()
execute_process(COMMAND git -C "${lace_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE lace_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT lace_revision STREQUAL PNDD_LACE_COMMIT)
  message(FATAL_ERROR "Lace source must be at ${PNDD_LACE_COMMIT}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_lace.py"
  "${lace_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lace-adapter" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/lace-adapter" "${lace_BINARY_DIR}")
if(PNDD_ROOT_AFFINITY)
  target_compile_definitions(lace PRIVATE PNDD_ROOT_AFFINITY=1)
  target_compile_definitions(lace14 PRIVATE PNDD_ROOT_AFFINITY=1)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(lace PRIVATE _GNU_SOURCE)
  target_compile_definitions(lace14 PRIVATE _GNU_SOURCE)
endif()
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/sylvan-adapter" sylvan-build)
target_compile_definitions(sylvan PUBLIC ZDD_COMPLEMENT_EDGES=0)
target_compile_options(sylvan PRIVATE -Wno-pedantic)
find_package(Threads REQUIRED)
add_library(pndd STATIC src/pndd.c)
target_compile_definitions(pndd PRIVATE PNDD_SYLVAN_INITIAL_RATIO=${PNDD_SYLVAN_INITIAL_RATIO})
foreach(feature PROFILE PAR_PROFILE OPTIMISTIC_CACHE BATCH_GRAIN PENDING_PARTITION LABEL_GRAIN FAST_BCDD SPLIT_CACHE PAD_ROOTS LOCAL_CACHE LEAF_APPLY RESIDUAL_APPLY LABEL_GC_KEEP_NDD NODE_SLABS COST_GRAIN EXPERIMENT_SATCOUNT_CACHE EXPERIMENT_BCDD_AND EXPERIMENT_SYLVAN_CACHE)
  if(PNDD_${feature})
    target_compile_definitions(pndd PRIVATE PNDD_${feature}=1)
  endif()
endforeach()
target_include_directories(pndd PUBLIC include)
target_link_libraries(pndd PUBLIC sylvan Threads::Threads m)
target_compile_options(pndd PRIVATE -Wall -Wextra)
if(PNDD_BUILD_JNI)
  find_package(JNI REQUIRED COMPONENTS JVM)
  add_library(pnddjni SHARED src/jni.c)
  target_include_directories(pnddjni PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(pnddjni PRIVATE pndd)
endif()
add_executable(pndd_entry bench/entry.c)
target_link_libraries(pndd_entry PRIVATE pndd)
add_executable(pndd_nqueens bench/nqueens.c)
target_link_libraries(pndd_nqueens PRIVATE pndd)
if(BUILD_TESTING)
  add_executable(pndd_test tests/test_pndd.c)
  target_compile_options(pndd_test PRIVATE -UNDEBUG)
  target_link_libraries(pndd_test PRIVATE pndd)
  foreach(workers 1 2 4 6)
    add_test(NAME pndd_correctness_${workers} COMMAND pndd_test ${workers})
    set_tests_properties(pndd_correctness_${workers} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
  endforeach()
  add_test(NAME pndd_cache_collision_6 COMMAND pndd_test 6 64)
  set_tests_properties(pndd_cache_collision_6 PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()

# MTNDD shares the manager source and production optimizations; only terminals
# and arithmetic are compiled additionally. Separate ABI/library from Boolean.
add_library(pmtndd STATIC src/pndd.c)
get_target_property(pmt_features pndd COMPILE_DEFINITIONS)
target_compile_definitions(pmtndd PRIVATE ${pmt_features} PNDD_MULTITERMINAL=1)
target_include_directories(pmtndd PUBLIC include)
target_link_libraries(pmtndd PUBLIC sylvan Threads::Threads m)
target_compile_options(pmtndd PRIVATE -Wall -Wextra)
if(PNDD_BUILD_JNI)
  add_library(pmtnddjni SHARED src/pmtndd_jni.c)
  target_include_directories(pmtnddjni PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(pmtnddjni PRIVATE pmtndd)
endif()
if(BUILD_TESTING)
  add_executable(pmtndd_test tests/test_pmtndd.c)
  target_compile_options(pmtndd_test PRIVATE -UNDEBUG)
  target_link_libraries(pmtndd_test PRIVATE pmtndd)
  foreach(workers 1 6)
    add_test(NAME pmtndd_correctness_${workers} COMMAND pmtndd_test ${workers})
    set_tests_properties(pmtndd_correctness_${workers} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
  endforeach()
endif()
