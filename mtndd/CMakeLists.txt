cmake_minimum_required(VERSION 3.23)
project(parallel_mtndd LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(MTNDD_SANITIZE "Enable address/undefined sanitizers for Debug checks" OFF)
if(MTNDD_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(CTest)
include(FetchContent)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
# Reuse the Boolean branch's build-local Lace compatibility/liveness adapter.
set(LACE_ENABLE_PIC ON CACHE BOOL "" FORCE)
set(LACE_NATIVE_OPT OFF CACHE BOOL "Portable default" )
FetchContent_Declare(lace GIT_REPOSITORY https://github.com/trolando/lace.git
  GIT_TAG 63d8a5ea79922a234e284ee4b264fa5e4e784939)
FetchContent_GetProperties(lace)
if(NOT lace_POPULATED)
  FetchContent_Populate(lace)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_LIST_DIR}/../native/scripts/prepare_lace.py"
  "${lace_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lace-adapter"
  COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/lace-adapter" "${lace_BINARY_DIR}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(lace PRIVATE _GNU_SOURCE)
  target_compile_definitions(lace14 PRIVATE _GNU_SOURCE)
endif()
set(SYLVAN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SYLVAN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SYLVAN_ENABLE_PIC ON CACHE BOOL "" FORCE)
FetchContent_Declare(sylvan GIT_REPOSITORY https://github.com/Augists/sylvan.git
  GIT_TAG c8bf561ee64036375f4693eb389e57ec8ef4aa6d)
FetchContent_MakeAvailable(sylvan)
add_subdirectory(mtpndd)
option(MTPNDD_BUILD_JNI "Build MTNDD JNI bindings" ON)
if(MTPNDD_BUILD_JNI)
  add_subdirectory(jni)
endif()
